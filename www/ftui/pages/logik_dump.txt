      : ReadingsNum($FC,
          sprintf("pv_fc_tomorrow_%02d",$t - 24),0)
        * 1000;
  };
  my $lst = sub {
    my ($t) = @_;
    my $v = ReadingsNum($GP,
      sprintf("grundlast_h%02d_ohne_ev",$t % 24),0);
    $v = $LAST_MIN if($v < $LAST_MIN);
    return $v;

      sprintf("pv_fc_today_%02d",$h),0);

      sprintf("pv_begr_min_%02d",$h),0) > 0);
    $ist += ReadingsNum($FC,
      sprintf("pv_ist_%02d",$h),0) / 1000;

      : $fct->($t);

########## di_PV_Einspeiseleistung_anpassen (DOIF)
ATTR do: always
ATTR event-on-change-reading: .*,ramp_last_t
ATTR room: 01 PV
ATTR userReadings: ramp_last_t_hr:ramp_last_t.* {my $t=ReadingsNum("di_PV_Einspeiseleistung_anpassen","ramp_last_t",0); $t>0?FmtDateTime($t):"-"}
DEF:
([di_PV_Jahresbilanz:einsp_abweichung_kwh]
or [OpenWB_Garage_links:Ladeleistung_gesamt]
or [OpenWB_Garage_rechts:Ladeleistung_gesamt]
or [+00:01])
{
  my $now = time();
  my $DEV = "di_PV_Einspeiseleistung_anpassen";

  ## Parameter aus EMS_Control (mit Code-Defaults als Fallback)
  my $abw      = ReadingsNum("di_PV_Jahresbilanz","einsp_abweichung_kwh",0);
  my $lmin     = ReadingsNum("EMS_Control","dynLimitMinW",2000);
  my $lmax     = ReadingsNum("EMS_Control","dynLimitMaxW",15000);
  my $alow     = ReadingsNum("EMS_Control","dynAbwLowKwh",500);
  my $ahigh    = ReadingsNum("EMS_Control","dynAbwHighKwh",2000);
  my $lev      = ReadingsNum("EMS_Control","dynLimitEvW",4000);
  my $EV_MIN_W = ReadingsNum("EMS_Control","dynEvMinW",700);
  my $RAMP_STEP= ReadingsNum("EMS_Control","dynRampStepW",2000);
  my $RAMP_INT = ReadingsNum("EMS_Control","dynRampIntSek",60);

  ## Laedt das E-Auto gerade?
  my $ev_power = ReadingsNum("OpenWB_Garage_links","Ladeleistung_gesamt",0) + ReadingsNum("OpenWB_Garage_rechts","Ladeleistung_gesamt",0);
  my $ev_laedt = ($ev_power >= $EV_MIN_W) ? 1 : 0;

  ## Ziel-Limit bestimmen
  my $ziel;
  my $grund;
  if ($ev_laedt) {
    $ziel  = $lev;
    $grund = "EV_laedt";
  }
  else {
    if ($abw <= $alow) { $ziel = $lmax; }
    elsif ($abw >= $ahigh) { $ziel = $lmin; }
    else {
      my $f = ($abw - $alow) / ($ahigh - $alow);
      $ziel = $lmax - $f * ($lmax - $lmin);
    }
    $ziel = int($ziel / 500 + 0.5) * 500;
    $ziel = $lmin if ($ziel < $lmin);
    $ziel = $lmax if ($ziel > $lmax);
    $grund = "Abweichung_".$abw."_kWh";
  }

  ## Sanftes Herunterrampen (nur runter, hoch sofort)
  my $alt    = ReadingsNum("EMS_Control","maxFeedInW5876",$lmax);
  my $ramp_t = ReadingsNum($DEV,"ramp_last_t",0);
  my $lim;
  if ($ziel >= $alt) {
    $lim = $ziel;
  }
  else {
    if (($now - $ramp_t) >= $RAMP_INT) {
      $lim = $alt - $RAMP_STEP;
      $lim = $ziel if ($lim < $ziel);
      fhem("setreading $DEV ramp_last_t ".int($now));
      fhem("setreading $DEV ramp_last_zeit ".FmtDateTime($now));
    }
    else {
      $lim = $alt;
    }
    $grund .= "_Rampe" if ($lim > $ziel);
  }

  ## Nur bei Aenderung an die Wechselrichter schreiben
  if ($lim != $alt) {
    fhem("set GoodWe_Haus_Ost_Mitte_5876 EMSFeedPowerEnable 1");
    fhem("set GoodWe_Garage_Haus_West_5877 EMSFeedPowerEnable 1");
    fhem("setreading EMS_Control maxFeedInW5876 $lim");
    fhem("setreading EMS_Control maxFeedInW5877 $lim");
    fhem("setreading openWB_Garage_links_Control aktivFeedLimitW $lim"); fhem("setreading openWB_Garage_rechts_Control aktivFeedLimitW $lim");
  }

  ## Status-Readings
  my $st = $ev_laedt ? "ev_offen"
         : ($lim > $ziel)  ? "rampe_abklingend"
         : ($lim >= $lmax) ? "offen"
         : ($lim <= $lmin) ? "max_begrenzt" : "dynamisch";
  fhem("setreading $DEV limit_aktiv ".$st."_".$lim."W");
  fhem("setreading $DEV limit_grund ".$grund."_Limit_".$lim."W");
  fhem("setreading $DEV limit_ziel ".int($ziel));
}

$wunsch = 1 if($power > 50); 
$sr->("bat_entl_wunsch",$wunsch); 
 
## ---------- manueller Stopp / Fremdmodus in openWB -> Finger weg 
if($manual) 
{ 
$sr->("active_mode","manual_".$cm); 
$sr->("plugged",$plug); 
return; 
} 
 
## ---------- kein EV gesteckt -> pv1 
if(!$plug) 
{ 
$setMode->("pv1"); 
$resetAll->(); 
$sr->("active_mode","unplugged"); 
return; 
} 
 
## ---------- EV neu gesteckt 
if(!$rn->("plugged",0)) 
{ 
$resetAll->(); 
$sr->("plugged",1); 
$sr->("plug_t",int($now)); 
} 
 
## ---------- EV voll -> pv1 bis zum Abstecken 
if($rn->("car_full",0)) 
{ 
$setMode->("pv1"); 
$sr->("active_mode","car_full"); 
return; 
} 
 
if(($now - $rn->("plug_t",$now)) < $PLUG_WAIT) 
{ 
$sr->("active_mode","plug_in_wait"); 
return; 
} 
 
## ---------- PV-Mindestleistung (Hysterese + Einschaltverzoegerung) 
my $pv_ok = $rn->("pv_ok",0); 
if(!$pv_ok) 
{ 
if($pv >= $PV_ON) 
{ 
my $hs = $rn->("pv_high_since",0); 
if($hs == 0) 
{ 
$sr->("pv_high_since",int($now)); 
} 
elsif(($now - $hs) >= $PV_ON_SEC) 
{ 
$pv_ok = 1; 
$sr->("pv_ok",1); 
$sr->("pv_low_since",0); 
$sr->("pv_high_since",0); 
} 
} 
else 
{ 
$sr->("pv_high_since",0); 
} 
} 
else 
{ 
if($pv < $PV_ON - $PV_HYST) 
{ 
my $ls = $rn->("pv_low_since",0); 
if($ls == 0) 
{ 
$sr->("pv_low_since",int($now)); 
} 
elsif(($now - $ls) >= $PV_DELAY) 
{ 
$pv_ok = 0; 
$sr->("pv_ok",0); 
$sr->("pv_low_since",0); 
} 
} 
else 
{ 
$sr->("pv_low_since",0); 
} 
} 
 
if(!$pv_ok) 
{ 
$setMode->("pv1"); 
$sr->("phase_mode",1); 
$sr->("sofort_since",0); 
$sr->("p3_since",0); 
$sr->("bezug3_since",0); 
$sr->("low_since",0); 
$sr->("active_mode",($rn->("pv_high_since",0) > 0) ? "pv_start_verzoegerung" : "pv_zu_gering"); 
return; 
} 
 
## ---------- Vorrang rechts: Pause links, wenn fuer 6 A kein Platz bleibt 
my $ph0  = ($rn->("phase_mode",1) == 3) ? 3 : 1; 
my $minW = $MIN_A * 230 * $ph0; 
my $platz = $feed + $power - $ziel; 
my $prio_pause = $rn->("prio_pause",0); 
if($r_aktiv && !$prio_pause) 
{ 
if($platz < $minW && ($now - $rn->("sofort_since",$now)) >= $PRIO_PAUSE_SEC) 
{ 
my $s = $rn->("prio_low_since",0); 
if($s == 0) 
{ 
$sr->("prio_low_since",int($now)); 
} 
elsif(($now - $s) >= $PRIO_PAUSE_SEC) 
{ 
$prio_pause = 1; 
$sr->("prio_pause",1); 
$sr->("prio_low_since",0); 
} 
} 
else 
{ 
$sr->("prio_low_since",0); 
} 
} 
elsif($prio_pause) 
{ 
if(!$r_aktiv || ($feed - $ziel) >= $minW + $PRIO_RESUME_W) 
{ 
my $s = $rn->("prio_high_since",0); 
if($s == 0) 
{ 
$sr->("prio_high_since",int($now)); 
} 
elsif(($now - $s) >= $PRIO_PAUSE_SEC || !$r_aktiv) 
{ 
$prio_pause = 0; 
$sr->("prio_pause",0); 
$sr->("prio_high_since",0); 
} 
} 
else 
{ 
$sr->("prio_high_since",0); 
} 
} 
if($prio_pause) 
{ 
$setMode->("pv1"); 
$sr->("phase_mode",1); 
$sr->("sofort_since",0); 
$sr->("p3_since",0); 
$sr->("bezug3_since",0); 
$sr->("low_since",0); 
$sr->("active_mode","prio_rechts_pause"); 
return; 
} 
 
## ---------- Sofortladen aktiv halten 
my $ph   = ($rn->("phase_mode",1) == 3) ? 3 : 1; 
my $want = ($ph == 3) ? "sofortMax" : "sofort1"; 
my $maxA = ($ph == 3) ? $MAX_A3 : $MAX_A1; 
my $phP  = 230 * $ph; 
 
if($rn->("sofort_since",0) == 0) 
{ 
$sr->("sofort_since",int($now)); 
$setA->($MIN_A); 
readingsSingleUpdate($hash,"last_chg_t",int($now),0); 
readingsSingleUpdate($hash,"last_dn_t",int($now),0); 
} 
$setMode->($want); 
if($cm ne "instant_charging" 
&& ($now - $rn->("mode_sent_t",0)) >= $RESEND) 
{ 
$setMode->($want,1); 
} 
 
## ---------- Umschaltpause erkennen 
## Waehrend der Phasenumschaltung laedt das EV nicht -> Einspeisung springt hoch. 
## Bis das EV wieder laedt und SW_SETTLE s stabil laeuft: keine Hochregelung, 
## keine Rueckschaltung, keine Voll-Erkennung. 
my $sw_t  = $rn->("sw_t",0); 
my $in_switch = 0; 
if(($now - $sw_t) < $SW_WIN) 
{ 
if($power < $FULL_W) 
{ 
$in_switch = 1; 
$sr->("run_since",0); 
} 
else 
{ 
my $rs = $rn->("run_since",0); 
if($rs == 0) 
{ 
$sr->("run_since",int($now)); 
$rs = $now; 
} 
$in_switch = 1 if(($now - $rs) < $SW_SETTLE); 
} 
} 
$sr->("in_switch",$in_switch); 
 
## ---------- EV voll / Ladung unterbrochen 
if($power < $FULL_W && ($now - $sw_t) >= $SW_WIN) 
{ 
my $ls = $rn->("low_since",0); 
if($ls == 0) 
{ 
$sr->("low_since",int($now)); 
} 
elsif(($now - $ls) >= $FULL_SEC 
&& ($now - $rn->("sofort_since",$now)) >= $FULL_SEC) 
{ 
$sr->("car_full",1); 
$sr->("car_full_lock",1); 
$setMode->("pv1"); 
$sr->("active_mode","car_full"); 
return; 
} 
} 
else 
{ 
$sr->("low_since",0); 
} 
 
## ---------- Phasenumschaltung 
if($ph == 1) 
{ 
if($power >= $P3_W && $sw_count < $SW_MAX && !$r_switch) 
{ 
my $s = $rn->("p3_since",0); 
if($s == 0) 
{ 
$sr->("p3_since",int($now)); 
} 
elsif(($now - $s) >= $P3_SEC) 
{ 
my $a3 = POSIX::floor($power / (3 * 230)); 
$a3 = $MIN_A if($a3 < $MIN_A); 
$a3 = $MAX_A3 if($a3 > $MAX_A3); 
$sr->("phase_mode",3); 
$sr->("sw_count",$sw_count + 1); 
$sr->("sw_t",int($now)); 
$sr->("p3_since",0); 
$sr->("bezug3_since",0); 
$setA->($a3); 
$setMode->("sofortMax"); 
readingsSingleUpdate($hash,"last_chg_t",int($now),0); 
$sr->("active_mode","switch_1p_3p"); 
return; 
} 
} 
else 
{ 
$sr->("p3_since",0); 
} 
} 
else 
{ 
if(!$in_switch && $grid > $BEZUG_MIN) 
{ 
my $s = $rn->("bezug3_since",0); 
if($s == 0) 
{ 
$sr->("bezug3_since",int($now)); 
} 
elsif(($now - $s) >= $P1_SEC) 
{ 
my $a1 = POSIX::floor(($power + $feed - $ziel) / 230); 
$a1 = $MIN_A if($a1 < $MIN_A); 
$a1 = $MAX_A1 if($a1 > $MAX_A1); 
$sr->("phase_mode",1); 
$sr->("sw_count",$sw_count + 1); 
$sr->("sw_t",int($now)); 
$sr->("bezug3_since",0); 
$setA->($a1); 
$setMode->("sofort1"); 
readingsSingleUpdate($hash,"last_chg_t",int($now),0); 
$sr->("active_mode","switch_3p_1p"); 
return; 
} 
} 
else 
{ 
$sr->("bezug3_since",0); 
} 
} 
 
## ---------- Stromregelung 
my $cur = $rn->("targetA",$MIN_A); 
$cur = $MIN_A if($cur < $MIN_A); 
$cur = $maxA if($cur > $maxA); 
my $neu = $cur; 
my $am  = "regelung"; 
 
if($grid > $BEZUG_MIN) 
{ 
if(($now - $rn->("last_dn_t",0)) >= $DN_SEC) 
{ 
my $red = POSIX::ceil($grid * $BEZUG_F / $phP); 
$neu = $cur - $red; 
$neu = $MIN_A if($neu < $MIN_A); 
readingsSingleUpdate($hash,"last_dn_t",int($now),0) if($neu < $cur); 
} 
$am = ($neu < $cur) ? "bezug_cut" : "bezug_min"; 
} 
else 
{ 
my $diff = $feed - $ziel; 
## Hochregeln nur, wenn das EV den bisherigen Sollstrom auch zieht 
## (verhindert Hochlaufen waehrend Umschaltpause / Anlauf / EV-seitiger Begrenzung) 
my $follow = ($power >= $FOLLOW_F * $cur * $phP) ? 1 : 0; 
$sr->("ev_follow",$follow); 
if($in_switch || $r_switch) 
{ 
$am = $in_switch ? "switch_pause" : "rechts_switch_pause"; 
} 
elsif($diff >= $phP && $follow && ($now - $rn->("last_chg_t",0)) >= $UP_SEC) 
{ 
$neu = $cur + 1; 
$am = "up"; 
} 
elsif($diff <= -$phP && ($now - $rn->("last_chg_t",0)) >= $UP_SEC) 
{ 
$neu = $cur - 1; 
$am = "down"; 
} 
} 
$neu = $MIN_A if($neu < $MIN_A); 
$neu = $maxA if($neu > $maxA); 
if($neu != $cur) 
{ 
readingsSingleUpdate($hash,"last_chg_t",int($now),0); 
} 
$setA->($neu); 
$sr->("active_mode",$am."_".$ph."p"); 
return; 
} 
)

fhem("set $CMD $m");

$a = int($a);

$sr->("plugged",0); 
$sr->("car_full",0); 
$sr->("phase_mode",1); 
$sr->("targetA",0); 
$sr->("p3_since",0); 
$sr->("bezug3_since",0);

my $wunsch = ($plug && !$manual && ($power > 50 || $sofort_aktiv)) ? 1 : 0; 
$wunsch = 1 if($power > 50); 
$sr->("bat_entl_wunsch",$wunsch);

if($manual) 
{ 
$sr->("active_mode","manual_".$cm); 
$sr->("plugged",$plug); 
return; 
} 
 
## ---------- kein EV gesteckt -> pv1 
if(!$plug) 
{ 
$setMode->("pv1"); 
$resetAll->(); 
$sr->("active_mode","unplugged"); 
return; 
} 
 
## ---------- EV neu gesteckt 
if(!$rn->("plugged",0)) 
{ 
$resetAll->(); 
$sr->("plugged",1); 
$sr->("plug_t",int($now)); 
} 
 
## ---------- EV voll -> pv1 bis zum Abstecken 
if($rn->("car_full",0)) 
{ 
$setMode->("pv1"); 
$sr->("active_mode","car_full"); 
return; 
} 
 
if(($now - $rn->("plug_t",$now)) < $PLUG_WAIT) 
{ 
$sr->("active_mode","plug_in_wait");

## ---------- PV-Mindestleistung (Hysterese + Einschaltverzoegerung) + Wiederanlaufsperre nach Bezug-Stopp

my $stop_t = $rn->("stop_t",0);

{ 
my $need = $PV_ON; 
if($stop_t > 0) 
{ 
if($pv < $PV_ON - $PV_HYST) 
{ 
$sr->("stop_t",0); 
$sr->("pv_stop_w",0); 
$stop_t = 0; 
} 
else 
{ 
my $n2 = $rn->("pv_stop_w",0) + $MIN_A * 230; 
$need = $n2 if($n2 > $need); 
$need = 999999 if(($now - $stop_t) < $RST_SEC); 
} 
} 
$sr->("pv_start_need",($need >= 999999) ? "sperre" : int($need)); 
if($pv >= $need) 
{ 
my $hs = $rn->("pv_high_since",0); 
if($hs == 0) 
{ 
$sr->("pv_high_since",int($now)); 
} 
elsif(($now - $hs) >= $PV_ON_SEC) 
{ 
$pv_ok = 1; 
$sr->("pv_ok",1); 
$sr->("pv_low_since",0); 
$sr->("pv_high_since",0); 
$sr->("stop_t",0); 
$sr->("pv_stop_w",0); 
$stop_t = 0; 
} 
} 
else 
{ 
$sr->("pv_high_since",0); 
} 
} 
else 
{ 
if($pv < $PV_ON - $PV_HYST) 
{ 
my $ls = $rn->("pv_low_since",0); 
if($ls == 0) 
{ 
$sr->("pv_low_since",int($now)); 
} 
elsif(($now - $ls) >= $PV_DELAY) 
{ 
$pv_ok = 0; 
$sr->("pv_ok",0); 
$sr->("pv_low_since",0); 
} 
} 
else 
{ 
$sr->("pv_low_since",0); 
} 
} 
 
if(!$pv_ok) 
{ 
$setMode->("pv1"); 
$sr->("phase_mode",1); 
$sr->("sofort_since",0); 
$sr->("p3_since",0); 
$sr->("bezug3_since",0); 
$sr->("low_since",0); 
$sr->("bezug_stop_since",0); 
$sr->("active_mode",($stop_t > 0) ? "bezug_stop_warten" : ($rn->("pv_high_since",0) > 0) ? "pv_start_verzoegerung" : "pv_zu_gering"); 
return; 
} 
 
## ---------- Sofortladen aktiv halten 
my $ph   = ($rn->("phase_mode",1) == 3) ? 3 : 1; 
my $want = ($ph == 3) ? "sofortMax" : "sofort1"; 
my $maxA = ($ph == 3) ? $MAX_A3 : $MAX_A1; 
my $phP  = 230 * $ph; 
 
if($rn->("sofort_since",0) == 0) 
{ 
$sr->("sofort_since",int($now)); 
$setA->($MIN_A); 
readingsSingleUpdate($hash,"last_chg_t",int($now),0); 
readingsSingleUpdate($hash,"last_dn_t",int($now),0); 
} 
$setMode->($want); 
if($cm ne "instant_charging" 
&& ($now - $rn->("mode_sent_t",0)) >= $RESEND) 
{ 
$setMode->($want,1); 
} 
 
## ---------- Umschaltpause erkennen 
## Waehrend der Phasenumschaltung laedt das EV nicht -> Einspeisung springt hoch. 
## Bis das EV wieder laedt und SW_SETTLE s stabil laeuft: keine Hochregelung, 
## keine Rueckschaltung, keine Voll-Erkennung. 
my $sw_t  = $rn->("sw_t",0); 
my $in_switch = 0; 
if(($now - $sw_t) < $SW_WIN) 
{ 
if($power < $FULL_W) 
{ 
$in_switch = 1; 
$sr->("run_since",0); 
} 
else 
{ 
my $rs = $rn->("run_since",0); 
if($rs == 0) 
{ 
$sr->("run_since",int($now)); 
$rs = $now; 
} 
$in_switch = 1 if(($now - $rs) < $SW_SETTLE); 
} 
} 
$sr->("in_switch",$in_switch); 
 
## ---------- EV voll / Ladung unterbrochen 
if($power < $FULL_W && ($now - $sw_t) >= $SW_WIN) 
{ 
my $ls = $rn->("low_since",0); 
if($ls == 0) 
{ 
$sr->("low_since",int($now)); 
} 
elsif(($now - $ls) >= $FULL_SEC 
&& ($now - $rn->("sofort_since",$now)) >= $FULL_SEC) 
{ 
$sr->("car_full",1); 
$setMode->("pv1"); 
$sr->("active_mode","car_full"); 
return; 
} 
} 
else 
{ 
$sr->("low_since",0); 
} 
 
## ---------- Phasenumschaltung 
if($ph == 1) 
{ 
if($power >= $P3_W && $sw_count < $SW_MAX) 
{ 
my $s = $rn->("p3_since",0); 
if($s == 0) 
{ 
$sr->("p3_since",int($now)); 
} 
elsif(($now - $s) >= $P3_SEC) 
{ 
my $a3 = POSIX::floor($power / (3 * 230)); 
$a3 = $MIN_A if($a3 < $MIN_A); 
$a3 = $MAX_A3 if($a3 > $MAX_A3); 
$sr->("phase_mode",3); 
$sr->("sw_count",$sw_count + 1); 
$sr->("sw_t",int($now)); 
$sr->("p3_since",0); 
$sr->("bezug3_since",0); 
$setA->($a3); 
$setMode->("sofortMax"); 
readingsSingleUpdate($hash,"last_chg_t",int($now),0); 
$sr->("active_mode","switch_1p_3p"); 
return; 
} 
} 
else 
{ 
$sr->("p3_since",0); 
} 
} 
else 
{ 
if(!$in_switch && $grid > $BEZUG_MIN) 
{ 
my $s = $rn->("bezug3_since",0); 
if($s == 0) 
{ 
$sr->("bezug3_since",int($now)); 
} 
elsif(($now - $s) >= $P1_SEC) 
{ 
my $a1 = POSIX::floor(($power + $feed - $ziel) / 230); 
$a1 = $MIN_A if($a1 < $MIN_A); 
$a1 = $MAX_A1 if($a1 > $MAX_A1); 
$sr->("phase_mode",1); 
$sr->("sw_count",$sw_count + 1); 
$sr->("sw_t",int($now)); 
$sr->("bezug3_since",0); 
$setA->($a1); 
$setMode->("sofort1"); 
readingsSingleUpdate($hash,"last_chg_t",int($now),0); 
$sr->("active_mode","switch_3p_1p"); 
return; 
} 
} 
else 
{ 
$sr->("bezug3_since",0); 
} 
} 
 
## ---------- Stromregelung 
my $cur = $rn->("targetA",$MIN_A); 
$cur = $MIN_A if($cur < $MIN_A); 
$cur = $maxA if($cur > $maxA); 
my $neu = $cur; 
my $am  = "regelung"; 
 
if($grid > $BEZUG_MIN) 
{ 
if(($now - $rn->("last_dn_t",0)) >= $DN_SEC) 
{ 
my $red = POSIX::ceil($grid * $BEZUG_F / $phP); 
$neu = $cur - $red; 
$neu = $MIN_A if($neu < $MIN_A); 
readingsSingleUpdate($hash,"last_dn_t",int($now),0) if($neu < $cur); 
} 
$am = ($neu < $cur) ? "bezug_cut" : "bezug_min"; 
} 
else 
{ 
my $diff = $feed - $ziel; 
## Hochregeln nur, wenn das EV den bisherigen Sollstrom auch zieht 
## (verhindert Hochlaufen waehrend Umschaltpause / Anlauf / EV-seitiger Begrenzung) 
my $follow = ($power >= $FOLLOW_F * $cur * $phP) ? 1 : 0; 
$sr->("ev_follow",$follow); 
if($in_switch) 
{ 
$am = "switch_pause"; 
} 
elsif($diff >= $phP && $follow && ($now - $rn->("last_chg_t",0)) >= $UP_SEC) 
{ 
$neu = $cur + 1; 
$am = "up"; 
} 
elsif($diff <= -$phP && ($now - $rn->("last_chg_t",0)) >= $UP_SEC) 
{ 
$neu = $cur - 1; 
$am = "down"; 
} 
} 
$neu = $MIN_A if($neu < $MIN_A); 
$neu = $maxA if($neu > $maxA); 
 
if($ph == 1 && !$in_switch && $neu <= $MIN_A && $grid > $STOP_W) 
{ 
my $bs = $rn->("bezug_stop_since",0); 
if($bs == 0) 
{ 
$sr->("bezug_stop_since",int($now)); 
} 
elsif(($now - $bs) >= $STOP_SEC) 
{ 
$setMode->("pv1"); 
$sr->("pv_ok",0); 
$sr->("pv_low_since",0); 
$sr->("pv_stop_w",int($pv)); 
$sr->("stop_t",int($now)); 
$sr->("bezug_stop_since",0); 
$sr->("sofort_since",0); 
$sr->("low_since",0); 
$sr->("active_mode","bezug_stop"); 
return; 
} 
} 
else 
{ 
$sr->("bezug_stop_since",0); 
} 
 
if($neu != $cur) 
{ 
readingsSingleUpdate($hash,"last_chg_t",int($now),0); 
} 
$setA->($neu); 
$sr->("active_mode",$am."_".$ph."p"); 
return; 
}
